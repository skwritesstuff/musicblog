        { title: "Wiz Khalifa – No Sleep (2011)", id: "evMR4aZMpjM" },
        { title: "Skrillex – Rock That Body Remix (2011)", id: "Z13MDsCReE0" },
        { title: "Timeflies – Adderall & Red Bull (2011)", id: "11OguJCoXqQ" },
        { title: "Curren$y ft. Freddie Gibbs – Scottie Pippen (2011)", id: "5Al3VmPNzBE" }
      ];

      let currentIndex = 0;
      let isPlaying = false;
      let isMuted = false;

      const playBtn = document.getElementById('radio-play-btn');
      const playIcon = document.getElementById('radio-play-icon');
      const playText = document.getElementById('radio-play-text');
      const prevBtn = document.getElementById('radio-prev-btn');
      const nextBtn = document.getElementById('radio-next-btn');
      const selectEl = document.getElementById('radio-track-select');
      const statusDot = document.getElementById('radio-status-dot');
      const eqBars = document.getElementById('radio-eq');
      const timeDisplay = document.getElementById('radio-time-display');
      const muteBtn = document.getElementById('radio-mute-btn');
      const container = document.getElementById('yt-radio-container');

      function initTrack(index) {
        if (!container) return;
        currentIndex = (index + playlist.length) % playlist.length;
        if (selectEl) selectEl.value = currentIndex;
        const track = playlist[currentIndex];
        container.innerHTML = `<iframe id="yt-radio-iframe" src="https://www.youtube-nocookie.com/embed/${track.id}?autoplay=1&enablejsapi=1&origin=${encodeURIComponent(window.location.origin || '*')}" style="position: absolute; width: 1px; height: 1px; left: -9999px; border: 0;" allow="autoplay"></iframe>`;
        setPlayingState(true);
      }

      function setPlayingState(playing) {
        isPlaying = playing;
        if (playIcon) playIcon.textContent = playing ? "⏸" : "▶";
        if (playText) playText.textContent = playing ? "PAUSE RADIO" : "PLAY RETRO RADIO";
        if (statusDot) {
          statusDot.style.background = playing ? "#10b981" : "#f59e0b";
          statusDot.style.boxShadow = playing ? "0 0 10px #10b981" : "none";
        }
        if (eqBars) {
          eqBars.style.opacity = playing ? "1" : "0.35";
        }
        if (timeDisplay) {
          timeDisplay.textContent = playing ? "PLAYING STREAM" : "STREAM PAUSED";
        }
      }

      if (playBtn) {
        playBtn.addEventListener('click', () => {
          if (!isPlaying) {
            initTrack(currentIndex);
          } else {
            if (container) container.innerHTML = "";
            setPlayingState(false);
          }
        });
      }

      if (prevBtn) {
        prevBtn.addEventListener('click', () => initTrack(currentIndex - 1));
      }
      if (nextBtn) {
        nextBtn.addEventListener('click', () => initTrack(currentIndex + 1));
      }
      if (selectEl) {
        selectEl.addEventListener('change', (e) => initTrack(parseInt(e.target.value, 10)));
      }
      if (muteBtn) {
        muteBtn.addEventListener('click', () => {
          isMuted = !isMuted;
          muteBtn.textContent = isMuted ? "🔇" : "🔊";
          const iframe = document.getElementById('yt-radio-iframe');
          if (iframe && iframe.contentWindow) {
            iframe.contentWindow.postMessage(JSON.stringify({
              event: 'command',
              func: isMuted ? 'mute' : 'unMute'
            }), '*');
          }
        });
      }
    })();
  </script>
